\subsection*{Problem 1.}
{\color {Gray}
Suppose $x$ and $y$ are real numbers. Find the minimum values of
\[
u = x^2 + 2xy + 5y^2 + 2x - 6y + 7.
\]
}
\MyLine
\begin{solution}
Because there exists $x^2, y^2, xy, x, y$, it is natural
to immediately think of the factorization form $(x \pm y + C)^2$.
\begin{align*}
    (x + y + 1)^2 + 4y^2 - 8y + 6 &= 0 \\
    (x + y + 1)^2 + 4(y - 1)^2 + 2 &= 0
\end{align*}
Thus, the minimum value of $u$ is $\boxed{2}$.
\end{solution}

\subsection*{Problem 2.}
{\color {Gray}
Find the maximum value of
\[
y = 2x - 5 + \sqrt{15 - 4x}
\]
for $x \in \mathbb{R}$.
}

\MyLine
\begin{solution}
Let $X = \sqrt{15 - 4x}$. Therefore, the maximum value
of $-\frac{X^2}{2} + X + \frac{5}{2}$ must be found.
\begin{align*}
    -\frac{X^2}{2} + X + \frac{5}{2}
    &= -\frac{1}{2} \left( X^2 -2X + 1 \right) + 3 \\
    &= -\frac{1}{2} \left( X - 1 \right)^2 + 3
\end{align*}
Because $X$ is a positive number, the maximum value of
$y$ is $\boxed{3}$.
\end{solution}

\subsection*{Problem 3.}
{\color {Gray}
Suppose $\frac{1}{2} \leq x^2 + 4y^2 \leq 2$ is true for
real numbers $x$ and $y$.
\begin{enumerate}
    \item Find the greatest value of $x^2 - 2xy + 4y^2$
    \item Find the smallest value of $x^2 - 2xy + 4y^2$
\end{enumerate}
}
\MyLine
\subsection*{Part 1.}
\textbf{Solution I.} Completing square method may provide
the maximum value.
\begin{align*}
    x^2 + 4y^2
    &= \frac{2}{3} \left( x^2 - 2xy + 4y^2 \right)
        + \frac{1}{3} \left( x^2 + 4xy + 4y^2 \right) \\
    &= \frac{2}{3} \left( x^2 - 2xy + 4y^2 \right)
        + \frac{1}{3} (x + 2y)^2
\end{align*}
Continuing,
\[
\frac{1}{2} \leq \frac{2}{3} \left( x^2 - 2xy + 4y^2 \right)
    + \frac{1}{3} (x + 2y)^2 \leq 2
\]
Notice that when $x = -2y$, $x^2 - 2xy + 4y^2 \leq 3$.
However, other cases will lead to $x^2 - 2xy + 4y^2$ being
less than or equal to a value smaller than $3$. Therefore,
$x^2 - 2xy + 4y^2$ will reach the maximum $3$ when $x = -2y$,
which is the case that provides the maximum value.
\qed

\textbf{Solution II.} Using AM-GM inequality, it is evident
that $x^2 + 4y^2 \geq |4xy|$. In other words, $|2xy| \leq 1$.
Moreover,
\[
\frac{1}{2} - 2xy \leq x^2 - 2xy + 4y^2 \leq 2 - 2xy
\]
Consider the case when $x = -1$ and $y = \frac{1}{2}$.
$x^2 + 4y^2$ will reach its maximum while $xy$ will face
its minimum. Thus, $2 + 1$, or $\boxed{3}$ is the maximum value.
\qed

\subsection*{Part 2.}
\begin{solution}
Completing square method could be used.
\begin{align*}
    x^2 + 4y^2
    &= 2 \left( x^2 - 2xy + 4y^2 \right) - (x^2 - 4xy + 4y^2) \\
    &= 2 \left( x^2 - 2xy + 4y^2 \right) - (x - 2y)^2
\end{align*}
Therefore,
\[
\frac{1}{2}
\leq 2 \left( x^2 - 2xy + 4y^2 \right)
    - (x - 2y)^2
\leq 2
\]
Notice that $x^2 - 2xy + 4y^2$ will reach its minimum
\textit{iff} $x = 2y$. Thus, the minimum value of
$x^2 - 2xy + 4y^2$ is $\boxed{\frac{1}{4}}$.
\end{solution}

\subsection*{Problem 4.}
{\color {Gray}
Find
\[
\frac{1}{s_{\max}} + \frac{1}{s_{\min}}
\]
if $s = x^2 + y^2$
with real $x$ and $y$ satisfying
\[
4x^2 - 5xy + 4y^2 = 5.
\]
}
\MyLine
\begin{solution}
Using quadratic formula, $x$ could be written in terms of $y$.
\[
x = \frac{5y \pm \sqrt{80 - 39y^2}}{8}
\]
Therefore, $-\sqrt{\frac{80}{39}} \leq x \leq \frac{80}{39}$
and $-\sqrt{\frac{80}{39}} \leq y \leq \frac{80}{39}$ since
the equation is symmetric.
\begin{align*}
    s
    &= \left( \frac{5y \pm \sqrt{80 - 39y^2}}{8} \right)^2 + y^2 \\
    &= \frac{25y^2 + 80 - 39y^2 \pm 10y\sqrt{80 - 39y^2} + 64y^2}{64} \\
    &= \frac{50y^2 \pm 10y\sqrt{80 - 39y^2} + 80}{64}
\end{align*}
Using the range for $y$, the following inequalities could be derived.
\begin{align*}
    0 &\leq \sqrt{80 - 39y^2} \leq \sqrt{80} \\
    0 &\leq 10y\sqrt{80 - 39y^2}
        \leq 10 \sqrt{\frac{80}{39}} \cdot \sqrt{80} \\
    80 &\leq 50y^2 + 10y\sqrt{80 - 39y^2} + 80
        \leq \frac{800}{\sqrt{39}} + \frac{400}{39} + 80 \\
    \frac{80}{64} &\leq \frac{50y^2 + 10y\sqrt{80 - 39y^2} + 80}{64}
        \leq \frac{\frac{800}{\sqrt{39}} + \frac{400}{39} + 80}{64}
\end{align*}
Similarly,
\begin{align*}
    -\sqrt{80} &\leq -\sqrt{80 - 39y^2} \leq 0 \\
    -10 \sqrt{\frac{80}{39}} \cdot
        \sqrt{80} &\leq -10y\sqrt{80 - 39y^2} \leq 0 \\
    80 &\leq 50y^2 - 10y\sqrt{80 - 39y^2} + 80
        \leq \frac{800}{\sqrt{39}} + \frac{400}{39} + 80 \\
    \frac{80}{64} &\leq \frac{50y^2 - 10y\sqrt{80 - 39y^2} + 80}{64}
        \leq \frac{\frac{800}{\sqrt{39}} + \frac{400}{39} + 80}{64}
\end{align*}
Since $\frac{80}{64} \leq s \leq \frac{\frac{800}{\sqrt{39}}
+ \frac{400}{39} + 80}{64}$, the value could be computed.
\begin{align*}
    \frac{1}{s_{\max}} + \frac{1}{s_{\min}}
    &= \frac{1}{\frac{\frac{800}{\sqrt{39}} + \frac{400}{39} + 80}{64}} + \frac{1}{\frac{80}{64}} \\
    &= \boxed{\frac{64}{\frac{800}{\sqrt{39}} + \frac{400}{39} + 80} + \frac{64}{80}}
\end{align*}
\end{solution}

\subsection*{Problem 5.}
{\color {Gray}
Find the greatest value of
\[
x^2 + y^2
\]
if $x$ and $y$ are real numbers satisfying
\[
3x^2 + 2y^2 = 2x.
\]
}
\MyLine
\begin{solution}
$y$ could be written in terms of $x$.
\[
y^2 = \frac{2x - 3x^2}{2}
\]
Because $y$ is a real number, $2x - 3x^2 \geq 0$,
or $0 \leq x \leq \frac{2}{3}$. Thus, the range could
be utilized.
\begin{align*}
    x^2 + y^2
    &= x^2 + \frac{2x - 3x^2}{2} \\
    &= \frac{-x^2 + 2x}{2}
    \\\\
    0 &\leq x \leq \frac{2}{3} \\
    -\frac{4}{9} &\leq x^2 \leq 0 \\
    -\frac{2}{9} &\leq \frac{x^2 + 2x}{2} \leq \frac{1}{3}
\end{align*}
Thus, the maximum value is $\boxed{\frac{1}{3}}$.
\end{solution}

\subsection*{Problem 6.}
{\color {Gray}
Let $x$ be any real number. Find the smallest value of
\[
\frac{3x^2 + 6x + 5}{\frac{1}{2}x^2 + x + 1}.
\]
}
\begin{solution}
\begin{align*}
    \frac{3x^2 + 6x + 5}{\frac{1}{2}x^2 + x + 1}
    &= \frac{3(x + 1)^2 + 2}{\frac{1}{2}(x + 1)^2 + \frac{1}{2}} \\
    &= \frac{6(x + 1)^2 + 4}{(x + 1)^2 + 1} \\
    &= 6 - \frac{2}{(x + 1)^2 + 1} \\
\end{align*}
Therefore, the minimum value $\boxed{4}$ is achieved when $x = -1$.
\end{solution}

\subsection*{Problem 7.}
{\color {Gray}
For $a, b \in \mathbb{R}$, find $a^2 + b^2$ if
\[
a\sqrt{1 - b^2} + b\sqrt{1 - a^2} = 1.
\]
}
\begin{solution}
One possible solution is when $a = 0$ and $b = 1$.
Thus, $a^2 + b^2 = \boxed{1}$.
\end{solution}
